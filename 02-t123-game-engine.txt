function permutations(items) {
  if (items.length === 1) return [items];
  return items.flatMap((item, index) =>
    permutations(items.filter((_, rest) => rest !== index)).map(rest => [item, ...rest]));
}

const assignments = permutations(Object.keys(MARKETS));

function makeRoom(start, assignmentIndex, suffix) {
  return "OMF-" + Math.floor(start / MINUTE).toString(36).toUpperCase() + "-" +
    assignmentIndex.toString(36).toUpperCase() + suffix.toUpperCase();
}

function roomParts(value) {
  return /^OMF-([A-Z0-9]{5,7})-([0-9A-N])([A-Z0-9]{3})$/.exec(String(value).trim().toUpperCase());
}

function roomStart(value) {
  const match = roomParts(value);
  if (!match) return null;
  const start = parseInt(match[1], 36) * MINUTE;
  return start >= Date.UTC(2025, 0, 1) && start <= Date.UTC(2100, 0, 1) ? start : null;
}

function marketForTable(room, team) {
  const match = roomParts(room);
  return match ? assignments[parseInt(match[2], 36)]?.[team - 1] : null;
}

function phaseAt(room, now) {
  const minute = (now - roomStart(room)) / MINUTE;
  const phase = SCHEDULE.find(item => minute >= item.from && minute < item.to);
  return phase ? {...phase, minute, bulletinB: phase.id === "R6" && minute >= 92} :
    {id: minute < 0 ? "SETUP" : "END", minute, to: minute < 0 ? 0 : null, bulletinB: false};
}

function endOfRound(room, id) {
  return roomStart(room) + SCHEDULE.find(phase => phase.id === id).to * MINUTE;
}

function startOfRound(room, id) {
  return roomStart(room) + SCHEDULE.find(phase => phase.id === id).from * MINUTE;
}

function initialProgress() {
  return {
    credits: CONFIG.budget, profile: null, research: null, route: null, verified: false, narrow: false,
    cohorts: 1, diligence: false, incident: null, launchEvidence: false, result: null
  };
}

function option(id, cost, label, description, blocked = "") {
  return {id, cost, label, description, blocked};
}

function optionsFor(id, state, marketId) {
  const market = MARKETS[marketId];
  const profile = market.profiles[state.profile];
  let options;
  switch (id) {
    case "R1":
      options = Object.entries(market.profiles).map(([key, item]) => option(key, 0, item.name, item.journey));
      break;
    case "R2":
      options = [
        option("desk", 0, "Keep the base dossier", "Use the market facts and first customer signal. Preserve credits for entry or the emergency."),
        option("interview", PRICE.interview, "Buy an interview follow-up", "Get evidence about why this segment might switch. This measures stated preferences."),
        option("prototype", PRICE.prototype, "Test the mock journey", "Get a result on how users interpret the quote, status or support screen. No money moves.")
      ];
      break;
    case "R3":
      options = [
        option("pilot", PRICE.pilot, ROUTES.pilot, "Scope one use case, in one city, with one cohort. The local provider handles any authorized activity after permission and readiness checks."),
        option("alliance", PRICE.alliance, ROUTES.alliance, "Prepare a co-branded offer and define distribution and support responsibilities. In R5 you may add a second cohort if you can afford it."),
        option("acquisition", PRICE.acquisition, ROUTES.acquisition, "Screen the target, plan diligence and integration. The deeper transaction check in R5 is needed for a deal mandate at the Board.")
      ];
      break;
    case "R4":
      options = [
        option("reserve", 0, "Keep a reserve", "Record a permission or contract dependency. Pay the standard rates if you later need to issue and test a correction."),
        option("verify", PRICE.verify, "Verify readiness", "Buy a scoped finding and pre-agreed correction wording. An approved one-channel correction is free. A full correction and retest costs one credit less.")
      ];
      break;
    case "R5":
      options = [
        option("keep", 0, "Keep your proposition", "Keep the use case and one channel. Defend a claim the competitor still leaves open."),
        option("narrow", PRICE.narrow, "Narrow and test the proposition", "Keep the R1 profile. Limit the corridor, subgroup or promise and unlock a five-person mock result. You may reuse the tested scope for a targeted correction."),
        option("broaden", PRICE.broaden, "Add a second cohort and channel", "Reach a second group in the same city and segment. Full correction costs more when two channels are exposed.",
          state.route === "alliance" ? "" : "Available on the alliance route."),
        option("diligence", PRICE.diligence, "Investigate the deal’s main risk", "Get a target-specific finding and a closing condition. This qualifies you to request deal work once outreach has been corrected.",
          state.route === "acquisition" ? "" : "Available on the acquisition route.")
      ];
      break;
    case "R6":
      options = [
        option("hold", 0, "Hold outreach", "Publish a holding statement and document the incident. The next ask can fund preparation and a later review."),
        option("targeted", state.verified ? 0 : PRICE.targeted, "Correct one channel", "Correct one channel and pause all uncorrected messages. Keep one cohort. Readiness verification or the narrow mock test lets you validate the correction in that cohort."),
        option("full", (state.cohorts === 2 ? PRICE.fullTwo : PRICE.fullOne) - Number(state.verified),
          "Correct every channel and retest", "Correct all planned channels and run a mock comprehension check with 20 people per cohort. Each cohort needs at least 16 correct responses. A prior prototype, preparation of approved wording or a narrowed test can improve the result.")
      ];
      break;
    case "R7": {
      const pilotBlock = !["pilot", "alliance"].includes(state.route) ? "Requires a pilot or alliance route." :
        state.launchEvidence ? "" : "Requires evidence that users understood the corrected offer.";
      const dealBlock = state.route !== "acquisition" ? "Requires the acquisition route." :
        !state.diligence ? "Requires the R5 deal-risk check." :
        state.incident === "hold" ? "Requires at least a targeted public correction in R6." : "";
      options = [
        option("pilot", 0, "Request conditional approval for a pilot", "The Board can approve the next step. Live activity still depends on the stated launch conditions.", pilotBlock),
        option("deal", 0, "Request a conditional mandate for deal work", "Request deeper diligence or negotiation against a specific closing condition. A go-live decision follows separately.", dealBlock),
        option("retest", 0, "Request one revised mock test", "Use one cohort with no live funds. Return to the Board before seeking pilot approval.",
          state.incident === "hold" ? "Outreach is on hold. Request preparation and a review instead." : ""),
        option("defer", 0, "Defer live activity and fund preparation", "Request one preparatory step and set a condition and a date for returning to the Board.")
      ];
      break;
    }
    default: throw new Error("Unknown round.");
  }

  return options.map(item => ({
    ...item,
    blocked: item.blocked || (item.cost > state.credits ? "Needs " + item.cost + " credits. You have " + state.credits + "." : "")
  }));
}

function fallbackFor(round, state, marketId) {
  if (round.id === "R3") return option("planning", 0, ROUTES.planning, "Continue with one mock cohort. There is no live entry route to present.");
  return optionsFor(round.id, state, marketId).find(item => item.id === round.fallback);
}

function dealClause(route) {
  return {
    pilot: "Check that the local provider is ready and permitted to run the specified activity. The provider must approve claims about fees and availability.",
    alliance: "Set the exact approval responsibility in the draft agreement. Both firms must clear a co-branded public offer before release.",
    acquisition: "Check which permissions, vendor terms and claims would survive a change of control. The target’s existing arrangements do not carry over automatically."
  }[route] || "Scope a mock test and identify a licensed provider before discussing any live activity.";
}

function pendingFact(round, state, marketId) {
  const market = MARKETS[marketId];
  const profile = market.profiles[state.profile];
  return {
    R2: profile?.signal,
    R4: dealClause(state.route),
    R5: profile?.competitor,
    R7: "The Board can fund one next step and reserve first-launch priority for that market. State exactly what it is being asked to approve."
  }[round];
}

function responseResult(state, selection, profile) {
  if (selection === "hold") return {
    incident: "hold", launchEvidence: false, cohorts: state.cohorts,
    result: "Outreach stopped. Preserve the evidence and propose a review date."
  };
  if (selection === "targeted") {
    const loss = state.cohorts === 2 ? "The second cohort is removed. " : "";
    const proven = state.verified || state.narrow;
    const reason = state.verified ? "The prior wording review supports a scoped follow-up." :
      state.narrow ? "The earlier narrow prototype supports a scoped follow-up." : "";
    return {
      cohorts: 1, incident: "targeted", launchEvidence: proven,
      result: loss + (proven ? reason + " Four of five mock users in the retained cohort understood the correction. The result supports a conditional proposal for that scope." :
        "One channel is corrected. No comprehension result is available yet. A new test is needed before a pilot ask.")
    };
  }
  const advantage = state.narrow ? 2 : Number(state.verified || state.research === "prototype");
  const firstResult = state.narrow ? 18 : state.cohorts === 2 ? 16 + advantage : 15 + advantage;
  const counts = state.cohorts === 2 ? [firstResult, 17] : [firstResult];
  const launchEvidence = counts.every(count => count >= 16);
  return {
    incident: "full", launchEvidence, cohorts: state.cohorts,
    result: "Full retest: " + counts.map((count, index) => "cohort " + (index + 1) + " " + count + "/20").join(", ") +
      " correctly identified the scope and corrected claim. The threshold is 16/20 in each cohort. " +
      (launchEvidence ? "The retest met the threshold. " : "The retest missed the threshold. A revised test is needed for a pilot. ") + profile.incident
  };
}

function applyChoice(state, round, selection, marketId) {
  const market = MARKETS[marketId];
  const profile = market.profiles[state.profile];
  const next = {...state, credits: state.credits - selection.cost, result: null};
  switch (round.id) {
    case "R1":
      next.profile = selection.id;
      next.result = "Selected " + market.profiles[selection.id].name + ". The dossier and later events follow this profile.";
      break;
    case "R2":
      next.research = selection.id;
      next.result = {desk: "The base dossier is your evidence. Record what you would test next.", interview: profile.interview, prototype: profile.prototype}[selection.id];
      break;
    case "R3":
      next.route = selection.id;
      next.result = market.candidates[selection.id]?.join(" · ") || selection.description;
      break;
    case "R4":
      next.verified = selection.id === "verify";
      next.result = next.verified ? "Scoped review: " + profile.review + " Use the approved template: ‘This is a concept test for [scope]. Availability, timing and fees depend on [conditions]. For questions: [approved contact].’ This supports a one-channel follow-up and lowers the cost of the full retest." :
        "Reserve retained. The open condition remains: " + dealClause(state.route);
      break;
    case "R5":
      next.narrow = selection.id === "narrow";
      next.diligence = selection.id === "diligence";
      next.cohorts = selection.id === "broaden" ? 2 : 1;
      next.result = {keep: "One cohort. The key claim still needs to stand up to the rival’s offer.", narrow: profile.narrow,
        broaden: "Two cohorts and two channels. A targeted correction retains one cohort. A full retest can keep both.", diligence: market.diligence}[selection.id];
      break;
    case "R6":
      Object.assign(next, responseResult(state, selection.id, profile));
      break;
    case "R7":
      next.result = OUTCOMES[selection.id][1];
      break;
  }
  return next;
}

function isDraftComplete(draft) {
  return !!draft?.choice && !!draft.evidence?.trim() && !!draft.tradeoff?.trim();
}

function resolveChoice(event, round, state, marketId) {
  if (event.method === "default") {
    const fallback = fallbackFor(round, state, marketId);
    if (event.choice !== fallback.id) throw new Error("Unexpected default in " + round.id + ".");
    return fallback;
  }
  if (!["submitted", "deadline"].includes(event.method) || !isDraftComplete(event)) {
    throw new Error("Incomplete decision in " + round.id + ".");
  }
  const chosen = optionsFor(round.id, state, marketId).find(item => item.id === event.choice);
  if (!chosen || chosen.blocked) throw new Error(chosen?.blocked || "Invalid choice in " + round.id + ".");
  return chosen;
}

function replay(record) {
  let state = initialProgress();
  const path = record.history.map((event, index) => {
    const round = ROUNDS[index];
    if (!round || event.round !== round.id) throw new Error("Decisions are out of order.");
    const chosen = resolveChoice(event, round, state, record.market);
    state = applyChoice(state, round, chosen, record.market);
    return {...event, label: chosen.label, cost: chosen.cost, credits: state.credits, result: state.result, state};
  });
  return {state, path};
}

function newDecision(record, progress, round, method, now) {
  const draft = record.drafts[round.id] || {};
  const choice = method === "default" ? fallbackFor(round, progress, record.market).id : draft.choice;
  const event = {round: round.id, method, choice, at: new Date(now).toISOString(), evidence: draft.evidence || "", tradeoff: draft.tradeoff || ""};
  if (method !== "default") resolveChoice(event, round, progress, record.market);
  return {...record, history: [...record.history, event]};
}

function settleDeadlines(record, now) {
  if (record.mode !== "live") return record;
  for (const round of ROUNDS.slice(record.history.length)) {
    const deadline = endOfRound(record.room, round.id);
    if (now < deadline) break;
    const {state} = replay(record);
    const draft = record.drafts[round.id];
    const valid = isDraftComplete(draft) && optionsFor(round.id, state, record.market).some(item => item.id === draft.choice && !item.blocked);
    record = newDecision(record, state, round, valid ? "deadline" : "default", deadline);
  }
  return record;
}

function lockRound(record, roundId, now, useDefault = false) {
  const round = ROUNDS[record.history.length];
  if (round?.id !== roundId) throw new Error("This round is already closed.");
  assertOpen(record, roundId, now);
  if (roundId === "R6" && !canReadBulletinB(record, now)) throw new Error("Wait for bulletin B before locking a response.");
  if (useDefault && record.mode === "live") throw new Error("The classroom default applies when the round closes.");
  return newDecision(record, replay(record).state, round, useDefault ? "default" : "submitted", now);
}

function canReadBulletinB(record, now) {
  return record.mode === "practice" ? record.bulletinB :
    now >= roomStart(record.room) + 92 * MINUTE;
}

function assertOpen(record, roundId, now) {
  const round = ROUNDS.find(item => item.id === roundId);
  if (!round) throw new Error("Unknown round.");
  if (record.mode === "live") {
    if (phaseAt(record.room, now).id !== roundId) throw new Error("This round is not open.");
  } else {
    const active = ROUNDS[record.history.length]?.id;
    const justLocked = record.history.at(-1)?.round;
    if (active !== roundId && justLocked !== roundId) throw new Error("This practice round is closed.");
  }
}

function updateDraft(record, roundId, field, value, now) {
  assertOpen(record, roundId, now);
  if (ROUNDS[record.history.length]?.id !== roundId) throw new Error("The decision has already been locked.");
  if (field === "choice") {
    if (roundId === "R6" && !canReadBulletinB(record, now)) throw new Error("Wait for bulletin B.");
    const selected = optionsFor(roundId, replay(record).state, record.market).find(item => item.id === value && !item.blocked);
    if (!selected) throw new Error("This choice is unavailable.");
  } else if (field !== "evidence" && field !== "tradeoff") throw new Error("Unknown draft field.");
  const draft = {...record.drafts[roundId], [field]: value.slice(0, CONFIG.maxNote)};
  return {...record, drafts: {...record.drafts, [roundId]: draft}};
}

function updateNote(record, roundId, field, value, now) {
  assertOpen(record, roundId, now);
  if (!WORKSHEET[roundId]?.some(item => item[0] === field)) throw new Error("Unknown worksheet field.");
  return {...record, notes: {...record.notes, [roundId + "." + field]: value.slice(0, CONFIG.maxNote)}};
}

function updateBrief(record, value, now) {
  const ready = record.mode === "practice" ? record.history.length === ROUNDS.length : phaseAt(record.room, now).id === "PREP";
  if (!ready) throw new Error("The Board Brief is closed for editing.");
  return {...record, brief: value.slice(0, CONFIG.maxNote * 3)};
}

function normalizeNotes(source) {
  const notes = {};
  for (const [round, fields] of Object.entries(WORKSHEET)) {
    for (const [field] of fields) {
      const key = round + "." + field;
      if (typeof source?.[key] === "string") notes[key] = source[key].slice(0, CONFIG.maxNote);
    }
  }
  return notes;
}

function cleanText(value) {
  return typeof value === "string" ? value.slice(0, CONFIG.maxNote) : "";
}

function validateRecord(record) {
  if (record?.game !== CONFIG.game || record.version !== CONFIG.version) throw new Error("This file belongs to another game or rules version.");
  const {mode, team, market, room, history} = record;
  if (!["live", "practice"].includes(mode) || !TEAMS.includes(team) || !Object.hasOwn(MARKETS, market) ||
      (mode === "live" ? roomStart(room) === null || room !== room.trim().toUpperCase() || marketForTable(room, team) !== market : room !== "PRACTICE") ||
      !Array.isArray(history) || history.length > ROUNDS.length) throw new Error("Invalid team or room record.");
  const clean = {
    game: CONFIG.game, version: CONFIG.version, mode, team, market, room,
    session: cleanText(record.session), history: [], drafts: {},
    notes: normalizeNotes(record.notes), bulletinB: mode === "practice" && record.bulletinB === true,
    brief: typeof record.brief === "string" ? record.brief.slice(0, CONFIG.maxNote * 3) : ""
  };
  if (!clean.session) throw new Error("The record is missing a session ID.");
  if (mode === "practice" && history.length >= 6 && !clean.bulletinB) throw new Error("The emergency decision is missing bulletin B.");
  clean.history = history.map((event, index) => {
    const round = ROUNDS[index];
    const at = Date.parse(event?.at);
    if (!Number.isFinite(at)) throw new Error("Invalid timestamp in " + round.id + ".");
    if (mode === "live") {
      const first = startOfRound(room, round.id) + (round.id === "R6" && event.method === "submitted" ? 8 * MINUTE : 0);
      const last = endOfRound(room, round.id);
      const timely = event.method === "submitted" ? at >= first && at < last : at === last;
      if (!timely) throw new Error("Decision timestamp does not match the room schedule.");
    }
    return {round: event.round, choice: event.choice, method: event.method, at: new Date(at).toISOString(),
      evidence: cleanText(event.evidence), tradeoff: cleanText(event.tradeoff)};
  });
  replay(clean);
  for (const {id} of ROUNDS) {
    const draft = record.drafts?.[id];
    if (!draft || typeof draft !== "object") continue;
    clean.drafts[id] = {choice: cleanText(draft.choice), evidence: cleanText(draft.evidence), tradeoff: cleanText(draft.tradeoff)};
  }
  return clean;
}

function scoreJury(submissions) {
  if (!Array.isArray(submissions) || submissions.length !== 3 || ![0, 1, 2].every(index => submissions[index]?.complete)) return null;
  let written = 0;
  let oral = 0;
  let uppsala = 0;
  for (const expert of submissions) {
    RUBRIC.forEach(([section, name, weight], index) => {
      const checked = (expert.checks?.[index] || []).slice(0, 4).filter(check => check === true).length;
      const points = checked * weight / 4;
      if (section === "Written") written += points;
      else oral += points;
      if (name === "Uppsala and entry route") uppsala += points;
    });
  }
  return {written: written / 3, oral: oral / 3, uppsala: uppsala / 3, total: (written + oral) / 3};
}

function compareScores(a, b) {
  return b.total - a.total || b.oral - a.oral || b.uppsala - a.uppsala;
}

// src/storage.mjs
const prefix = CONFIG.game + ":" + CONFIG.version;
const memory = new Map();
let writable = true;

function storageAvailable() {
  return writable;
}

function getStored(key) {
  if (writable) {
    try {
      const value = localStorage.getItem(key);
      if (value === null) memory.delete(key);
      else memory.set(key, value);
      return value;
    } catch {
      writable = false;
    }
  }
  return memory.get(key) ?? null;
}

function putStored(key, value) {
  if (value === null) memory.delete(key);
  else memory.set(key, value);
  if (writable) {
    try {
      if (value === null) localStorage.removeItem(key);
      else localStorage.setItem(key, value);
    } catch {
      writable = false;
    }
  }
}

function teamKey(room, team) {
  return prefix + ":" + room + ":team:" + team;
}

function hostKey(room) {
  return prefix + ":" + room + ":facilitator";
}

function readSlot(room, team) {
  const raw = getStored(teamKey(room, team));
  if (raw === null) return null;
  const record = validateRecord(JSON.parse(raw));
  if (record.room !== room || record.team !== team) throw new Error("This saved record belongs to a different table.");
  return record;
}

function storeRecord(record) {
  putStored(teamKey(record.room, record.team), JSON.stringify(record));
}

function activeTeam(value) {
  const key = prefix + ":active";
  try {
    if (value === undefined) return JSON.parse(sessionStorage.getItem(key));
    if (value === null) sessionStorage.removeItem(key);
    else sessionStorage.setItem(key, JSON.stringify(value));
  } catch {
    return null;
  }
}

function newId() {
  return globalThis.crypto?.randomUUID?.() || Math.random().toString(36).slice(2) + Date.now().toString(36);
}

function newRecord(mode, room, team, market) {
  return validateRecord({
    game: CONFIG.game, version: CONFIG.version, mode, room, team, market,
    session: newId(), history: [], drafts: {}, notes: {}, brief: "", bulletinB: false
  });
}

function withSaveLock(key, change) {
  return navigator.locks?.request(key, change) || Promise.resolve().then(change);
}

function knownTable(value) {
  return value && TEAMS.includes(value.team) && (value.room === "PRACTICE" || roomStart(value.room) !== null);
}

function esc(value) {
  return String(value ?? "").replaceAll("&", "&amp;").replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;").replaceAll('"', "&quot;").replaceAll("'", "&#39;");
}

  return '<button type="button" class="pv-button' + (kind ? ' pv-button--' + kind : '') +

function saveFile(filename, data, mime) {
  const url = URL.createObjectURL(new Blob([data], {type: mime}));
  const link = document.createElement("a");
  link.href = url;
  link.download = filename;
  document.body.append(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

const moscowClock = new Intl.DateTimeFormat("en-GB", {timeZone: "Europe/Moscow", hour: "2-digit", minute: "2-digit", hourCycle: "h23"});

function interval(room, start, end) {
  const base = roomStart(room);
  return moscowClock.format(base + start * 60_000) + "–" + moscowClock.format(base + end * 60_000);
}

function countdown(delta) {
  const seconds = Math.max(0, Math.ceil(delta / 1000));
  return String(Math.floor(seconds / 60)).padStart(2, "0") + ":" + String(seconds % 60).padStart(2, "0");
}

function textReport(record, boardOnly = false) {
  const {state, path} = replay(record);
  const market = MARKETS[record.market];
  const profile = market.profiles[state.profile];
  const lines = [
    boardOnly ? "REVOLUT · BOARD BRIEF" : "REVOLUT · TEAM DECISION JOURNAL",
    "Table " + record.team + " · " + market.name + " · " + record.room,
    "Customer: " + (profile?.name || "Pending"),
    "Credits spent: " + (CONFIG.budget - state.credits) + " · remaining: " + state.credits
  ];
  if (path.length === ROUNDS.length) lines.push("Funding request: " + OUTCOMES[path.at(-1).choice][0]);
  if (record.brief) lines.push("", record.brief);
  if (!boardOnly) {
    for (const event of path) {
      lines.push("", event.round + " " + event.label + " · cost " + event.cost + " · balance " + event.credits + " · " + event.method);
      if (event.evidence) lines.push("Evidence: " + event.evidence);
      if (event.tradeoff) lines.push("Trade-off: " + event.tradeoff);
      if (event.result) lines.push("Result: " + event.result);
      for (const [field, label] of WORKSHEET[event.round]) {
        const value = record.notes[event.round + "." + field];
        if (value) lines.push(label + ": " + value);
      }
    }
  } else {
    if (!record.brief) lines.push("", "Team worksheet highlights:");
    const fields = [["R1.cage", "CAGE"], ["R3.later", "Next commitment"], ["R6.approval", "Reopening condition"], ["R7.board", "Board request"]];
    for (const [key, label] of fields) {
      if (!record.brief && record.notes[key]) lines.push(label + ": " + record.notes[key]);
    }
    lines.push("", "For the hearing: " + QUESTION);
  }
  return lines.join("\n");
}

function teamFilename(record, extension, name = "journal") {

  const time = host.start || (host.room ? moscowDate.format(roomStart(host.room)) + "T" + interval(host.room, 0, 1).slice(0, 5) : date + "T18:30");

  </div>`;
}

function scoresFor(team) {
  const data = host.panel.grades[team];

function scoresTable() {
  const scored = TEAMS.map(team => ({team, score: scoresFor(team)}));
  const ranked = scored.filter(item => item.score).sort((a, b) => compareScores(a.score, b.score));
  const complete = ranked.length === TEAMS.length;
  const tied = complete ? ranked.filter(item => compareScores(item.score, ranked[0].score) === 0) : [];
  const heading = !complete ? "Complete all three expert sheets for every table to rank the proposals." :
    tied.length > 1 ? "The Chair decides between tables " + tied.map(item => item.team).join(", ") + " and gives a rubric-based reason." :
      "First launch priority · Table " + ranked[0].team + " · " + MARKETS[marketForTable(host.room, ranked[0].team)].name;
  const rows = (complete ? ranked : scored).map(({team, score}) =>
    "<tr><td>" + esc(MARKETS[marketForTable(host.room, team)].name) + "</td>" +
    (score ? [score.written, score.oral, score.total].map(value => "<td>" + value.toFixed(1) + "</td>").join("") :
      '<td colspan="3">Pending</td>') + "</tr>").join("");
  return `<div class="pv-panel">
    <h3>Board ranking</h3><p>${esc(heading)}</p>
    <table class="pv-scores"><thead><tr><th scope="col">Market</th><th scope="col">Written</th><th scope="col">Oral</th><th scope="col">Total</th></tr></thead>
    <tbody>${rows}</tbody></table>
    <div class="pv-actions pv-file-actions">${button("Export scores CSV", "host-scores")}</div></div>`;
}

function hostClock(root, now) {
  const clock = root.querySelector("[data-host-clock]");
  if (!clock || !host.room) return;
  const phase = phaseAt(host.room, now);
  const bulletin = phase.id === "R6" && !phase.bulletinB;
  const delta = phase.to === null ? null : roomStart(host.room) + (bulletin ? 92 : phase.to) * 60_000 - now;
  const cue = {
    SETUP: "Create mixed tables, explain the mandate and run the practice example.",
    R1: "Release the Singapore baseline and assigned market. Remind teams to build their own CAGE.",
    R2: "The selected profile gets a customer signal and a choice of evidence packages.",
    R3: "Release local candidate files and ask for learning and commitment thresholds.",
    R4: "Explain what the scoped review buys. Save a JSON checkpoint before the break.",
    R5: "Each profile receives a different rival move. Ask how the team now competes.",
    R7: "Each team chooses what to fund and states the condition for its next commitment.",
    BREAK: "Export the checkpoint, then reopen the same room and table at the next seminar.",
    R6: bulletin ? BULLETINS.A : BULLETINS.B + " Each team sees its customer-specific note.",
    PREP: "Collect team JSON or written logs, finish the written checks and give teams the pitch order.",
    PITCH: "One four-minute slot per team. " + QUESTION,
    DEBRIEF: "Reconcile all scores. Announce the selected next investment and its live-launch condition.",
    END: "Session complete. Export the ranking and keep the team journals."
  }[phase.id] || phase.label;
  clock.dataset.near = delta !== null && delta <= 120_000;
  clock.querySelector("[data-host-phase]").textContent = phase.label || phase.id;
  clock.querySelector("[data-host-time]").textContent = delta === null ? "—" : countdown(delta);
  root.querySelector("[data-host-cue]").textContent = cue;
}

function exportScores() {
  const lines = ["table,market,written_70,oral_30,uppsala_16,total_100"];
  for (const team of TEAMS) {
    const score = scoresFor(team);
    lines.push([team, MARKETS[marketForTable(host.room, team)].name].concat(score ?
      [score.written, score.oral, score.uppsala, score.total] : ["","","",""]).join(","));
  }
  saveFile("revolut-" + host.room.toLowerCase() + "-scores.csv", lines.join("\n") + "\n", "text/csv;charset=utf-8");
}

async function handleHostAction(id, target, ui) {
  if (id === "host-create") {
    const start = target.closest(".pv-panel").querySelector('[data-host="start"]').value;
    const instant = Date.parse(start + "+03:00");
    if (!start || !Number.isFinite(instant)) throw new Error("Set a valid R1 start in Moscow time.");
    const random = crypto.getRandomValues(new Uint32Array(2));
    const next = makeRoom(instant, random[0] % 24, (random[1] % 46656).toString(36).padStart(3, "0"));
    if (roomStart(next) === null) throw new Error("Select a date between 2025 and 2099.");
    if (host.room && !await ui.ask("Switch to a new room?", "The current room’s exports can still be kept. The dashboard will open the new room.", "Create room")) return;
    host.room = next;
    host.start = start;
    host.selected = 1;
    host.panel = freshPanel();
    storePanel();
    ui.announce("Room created. Show the room code and table numbers to participants.");
    ui.render("#omf-stage-title");
  } else if (id === "host-scores") exportScores();
}

function updateHostField(input, render) {
  const {host: field} = input.dataset;
  if (field === "start") {
    host.start = input.value;
    return;
  }
  if (!host.room || !render) return;
  if (field === "selected") {
    host.selected = Number(input.value);
    render();
    return;
  }
  if (field !== "check" && field !== "complete") return;
  const team = host.selected;
  const expert = Number(input.dataset.expert);
  const grades = host.panel.grades[team] || [{checks: []}, {checks: []}, {checks: []}];
  const grade = grades[expert];
  if (field === "complete") grade.complete = input.checked;
  else {
    const row = Number(input.dataset.row);
    grade.checks[row] ||= [];
    grade.checks[row][Number(input.dataset.check)] = input.checked;
    grade.complete = false;
    input.closest("details").querySelector('[data-host="complete"]').checked = false;
  }
  input.closest("details").querySelector("summary").textContent = "Expert " + (expert + 1) + " · " + (grade.complete ? "complete" : "draft");
  host.panel.grades[team] = grades;
  storePanel();
  const previous = input.closest(".pv-stage-body").querySelector(".pv-scores").closest(".pv-panel");
  previous.outerHTML = scoresTable();
}

async function importTeamFile(file, ui) {
  if (!host.room) throw new Error("Create a room first.");
  if (file.size > CONFIG.maxImportBytes) throw new Error("The JSON file is too large.");
  let parsed;
  try { parsed = JSON.parse(await file.text()); }
  catch { throw new Error("Choose a JSON file exported by the game."); }
  const record = validateRecord(parsed);
  if (record.room !== host.room) throw new Error("This team file belongs to another room.");
  const current = host.panel.records[record.team];
  if (current && !await ui.ask("Replace the imported journal?", "Import the selected file for table " + record.team + "?", "Import")) return;
  host.panel.records[record.team] = record;
  host.selected = record.team;
  storePanel();
  ui.announce("Imported table " + record.team + " with " + record.history.length + "/7 locked decisions.");
  ui.render();
}
